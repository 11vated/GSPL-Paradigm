interface IdentityLayer {
  name: string
  archetype: "hero" | "villain" | "creature" | "deity"
  species: string
  tags: string[] // ["rabbit", "electric", "cosmic"]
  canonical?: boolean // is this based on real IP?

🔷 Morphology Layer (THIS CREATES SHAPE)
interface MorphologyLayer {
  symmetry: "bilateral" | "radial" | "asymmetric"

  proportions: {
    headToBody: number
    limbLength: number
    exaggeration: number // 0–1 (realistic → cartoon)
  }

  skeleton: {
    nodes: MorphNode[]
    connections: MorphConnection[]
  }
}

interface MorphNode {
  id: string
  type: "head" | "torso" | "limb" | "appendage"
  shape: "circle" | "oval" | "capsule" | "custom"

  position: [number, number]
}

interface MorphConnection {
  from: string
  to: string
  flexibility: number
}

🔷 Step 3: Morphology → Geometry
function generateGeometry(seed: GSPLSeed) {
  return seed.morphology.skeleton.nodes.map(node => {
    return createShape(node.shape, node.size, node.position)
  })
}
🔷 Step 4: Apply Constraints
function applyConstraints(geometry, constraints) {
  constraints.forEach(rule => {
    geometry = enforceRule(geometry, rule)
  })
  return geometry
}
🔷 Step 5: Rasterize → Sprite
function rasterize(geometry, style) {
  const canvas = createCanvas()

  drawShapes(canvas, geometry, style)
  applyShading(canvas, style)

  return canvasToSprite(canvas)
}
